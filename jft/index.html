  <script>

    // Kunci jawaban: pilihan nomor 1 (べんきょうします)

    const correctAnswer = 1;


    function checkAnswer(selectedOption, element) {

      const resultDiv = document.getElementById('result-message');

      const allButtons = document.querySelectorAll('.btn-option');


      // Nonaktifkan semua tombol setelah memilih

      allButtons.forEach(btn => btn.disabled = true);


      if (selectedOption === correctAnswer) {

        element.classList.add('correct');

        resultDiv.innerHTML = "⭕ 正解 (Benar)!";

        resultDiv.style.color = "#28a745";

      } else {

        element.classList.add('incorrect');

        // Tandai juga mana yang seharusnya benar

        allButtons[correctAnswer - 1].classList.add('correct');

        resultDiv.innerHTML = "❌ 不正解 (Salah)!";

        resultDiv.style.color = "#dc3545";

      }

    }

  </script>
